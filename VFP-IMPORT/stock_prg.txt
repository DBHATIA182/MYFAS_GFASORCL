 MFLDVAR='G_WEIGHT'
ELSE
 MFLDVAR='WEIGHT'

PROC STKLOT1   && FOR GOVIND KANSHI WITH RATE
PARA SDT,EDT,icode,SCODE,GODCODE,RPTORDER,MCP
MB='B'
MK='K'
MH='H'
ETYPE='R'
*
IF G_STK_TDG_WGT_TYPE='G'
 MFLDVAR='G_WEIGHT'
ELSE
 MFLDVAR='WEIGHT'
ENDI 
*
IF MCP='C'
 MDET=''
ELSE
 MDET="HAVING SUM(CASE WHEN E_TYPE=?ETYPE THEN NVL(QNTY,0) ELSE NVL(QNTY,0)*-1 END) <> 0"
ENDI 
MVAR="SELECT A.ITEM_CODE,A.LOT,max(A.B_NO) B_NO,A.SUP_CODE,B.ITEM_NAME,C.NAME,A.GOD_CODE,MIN(VR_dATE) VR_DATE,"+;
     "MAX(A.REMARKS) REMARKS,MAX(A.MSUP_CODE) MSUP_CODE,MAX(A.MSUP_NAME) MSUP_NAME,"+;
     "SUM(CASE WHEN E_TYPE='R' THEN NVL(QNTY,0) ELSE 0  END) QNTY,"+;
     "SUM(CASE WHEN E_TYPE='R' THEN NVL(&MFLDVAR,0) ELSE 0  END) WEIGHT,"+;
     "SUM(CASE WHEN E_TYPE='R' THEN NVL(AMOUNT,0) ELSE 0  END) AMOUNT,"+;
     "SUM(CASE WHEN STATUS='B' AND E_TYPE='R' THEN NVL(QNTY,0) WHEN STATUS='B' AND E_TYPE<>'R' THEN NVL(QNTY,0)*-1 END) BAGS,"+;
     "SUM(CASE WHEN STATUS='K' AND E_TYPE='R' THEN NVL(QNTY,0) WHEN STATUS='K' AND E_TYPE<>'R' THEN NVL(QNTY,0)*-1 END) KATTA,"+;
     "SUM(CASE WHEN STATUS='H' AND E_TYPE='R' THEN NVL(QNTY,0) WHEN STATUS='H' AND E_TYPE<>'R' THEN NVL(QNTY,0)*-1 END) HKATTA,"+;
     "SUM(CASE WHEN E_TYPE='R' THEN NVL(&MFLDVAR,0) ELSE NVL(&MFLDVAR,0)*-1 END) OWEIGHT "+; 
     "FROM LOTSTOCK A,ITEMMAST B,MASTER C WHERE A.COMP_CODE=?G_COMPCODE AND "+;
     "A.VR_DATE<=?EDT AND (A.COMP_CODE=B.COMP_CODE AND A.ITEM_CODE=B.ITEM_CODE) AND "+;
     "(A.COMP_CODE=C.COMP_CODE AND A.SUP_CODE=C.CODE) "+;
     "GROUP BY A.ITEM_CODE, A.LOT,A.SUP_CODE, B.ITEM_NAME, C.NAME, A.GOD_CODE &MDET"
    
If SQLExec(GNCONNHANDLE,Mvar,'X0') <=0
			                  =Aerror(laArr)
			                  Messagebox(MVAR+' '+laArr(2))
ENDIF
DO CASE
 CASE RPTORDER='B'
  MDET='B_NO,ITEM_CODE,LOT,VR_DATE'
 CASE RPTORDER='S'
  MDET='SUP_CODE,ITEM_CODE,LOT,VR_DATE' 
 CASE RPTORDER='D'
  MDET='VR_DATE,ITEM_CODE,LOT' 
 CASE RPTORDER='L'
  MDET='LOT,VR_DATE'
 CASE RPTORDER='R'
  MDET='REMARKS,VR_DATE' 
 OTHERWISE
  MDET='ITEM_CODE,LOT,VR_DATE'
  
ENDCASE  
SELECT ITEM_CODE,LOT,' ' STATUS,B_NO,SUP_CODE,ITEM_NAME,NAME,GOD_CODE,CAST(VR_DATE AS DATE) VR_DATE,;
NVL(QNTY,000000) QNTY,CAST(NVL(AMOUNT,0) AS NUMERIC(13,2)) AMOUNT,;
CAST(NVL(WEIGHT,0) AS NUMERIC(10,3)) WEIGHT,;
CAST(NVL(AMOUNT,0) AS NUMERIC(13,2))/CAST(NVL(WEIGHT,0) AS NUMERIC(10,3))*100 RATE,;
NVL(BAGS,000000) BAGS,NVL(KATTA,000000) KATTA,NVL(HKATTA,000000) HKATTA,;
CAST(OWEIGHT AS NUMERIC(13,3)) OWEIGHT,NVL(REMARKS,SPACE(25)) REMARKS,;
NVL(MSUP_CODE,SPACE(6)) MSUP_CODE,NVL(MSUP_NAME,SPACE(30)) MSUP_NAME FROM X0 ;
ORDER BY &MDET INTO CURSOR X1
MFILE='X1'
*
SELECT *,ROUND(NVL(WEIGHT,0)*NVL(RATE,0)/100,2) CL_AMT FROM &MFILE INTO CURSOR X2
MFILE='X2'
*
IF !EMPTY(SCODE)
 SELECT * FROM &MFILE WHERE SUP_CODE=SCODE ORDER BY &MDET INTO CURSOR X3
 MFILE='X3'
ENDI
IF ICODE<>0
 SELECT * FROM &MFILE WHERE ITEM_CODE=ICODE ORDER BY &MDET INTO CURSOR X4
 MFILE='X4'
ENDIF
*
IF !EMPTY(GODCODE)
 SELECT * FROM &MFILE WHERE GOD_CODE=GODCODE ORDER BY &MDET INTO CURSOR X5
ENDIF
