PROCEDURE refqry
MTYPE=THISFORM.TYPE.VALUE
MVAR='SELECT R_DATE,R_NO FROM PURCHASE A WHERE A.COMP_CODE=?G_COMPCODE AND A.TYPE=?MTYPE;
      GROUP BY R_dATE,R_NO ORDER BY 1,2'
SQLEXEC(GNCONNHANDLE,MVAR,'X0')
SELECT * FROM X0 INTO CURSOR FSEARCH
GO TOP
THISFORM.SEARCH()

ENDPROC
