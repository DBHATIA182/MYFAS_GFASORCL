
PARAMETERS tbname,fld1,fld2,FLD3,FLD4,FLD5,FILTVALUE,FILTFLD
IF !EMPTY(FILTVALUE) AND !EMPTY(FILTFLD)
 MVAR="SELECT &FLD1,&FLD2,&FLD3,&FLD4,&FLD5 FROM &TBNAME WHERE COMP_CODE=?G_COMPCODE AND &FILTFLD=?FILTVALUE ORDER BY 1"
ELSE
 MVAR="SELECT &FLD1,&FLD2,&FLD3,&FLD4,&FLD5 FROM &TBNAME WHERE COMP_CODE=?G_COMPCODE ORDER BY 1"
ENDI
If SQLExec(GNCONNHANDLE,Mvar,'X0') <=0
			                  =Aerror(laArr)
			                  Messagebox(MVAR+' '+laArr(2))
ENDIF
SELECT NVL(NAME,SPACE(40)) NAME,NVL(CITY,SPACE(20)) CITY,;
NVL(CODE,SPACE(6)) CODE,NVL(PAN,SPACE(10)) PAN,NVL(GST_NO,SPACE(15)) GST_NO;
FROM X0 INTO CURSOR F1HELP
DO FORM F1_HELP_PAN WITH 'F1HELP'  TO G_XX
